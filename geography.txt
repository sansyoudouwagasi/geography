        // --- SOUND EFFECTS ---
        function playSound(type) {
            if (!soundEnabled) return;
            try {
                const ctx = new (window.AudioContext || window.webkitAudioContext)();
                const osc = ctx.createOscillator();
                const gain = ctx.createGain();
                osc.connect(gain);
                gain.connect(ctx.destination);

                if (type === 'correct') {
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(523.25, ctx.currentTime);
                    osc.frequency.setValueAtTime(659.25, ctx.currentTime + 0.08);
                    osc.frequency.setValueAtTime(783.99, ctx.currentTime + 0.16);
                    gain.gain.setValueAtTime(0.2, ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.4);
                    osc.start();
                    osc.stop(ctx.currentTime + 0.4);
                } else if (type === 'wrong') {
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(220, ctx.currentTime);
                    osc.frequency.setValueAtTime(160, ctx.currentTime + 0.1);
                    gain.gain.setValueAtTime(0.2, ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.25);
                    osc.start();
                    osc.stop(ctx.currentTime + 0.25);
                } else if (type === 'levelup') {
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(440, ctx.currentTime);
                    osc.frequency.setValueAtTime(554.37, ctx.currentTime + 0.1);
                    osc.frequency.setValueAtTime(659.25, ctx.currentTime + 0.2);
                    gain.gain.setValueAtTime(0.3, ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.5);
                    osc.start();
                    osc.stop(ctx.currentTime + 0.5);
                }
            } catch(e) {}
        }

            } else {
                icon.className = 'fa-solid fa-volume-xmark text-xs text-slate-400';
            }
        }

        // --- LOCAL STORAGE DATA PERSISTENCE ---
        function loadSavedState() {
            try {
                const saved = localStorage.getItem('zengeos_save_v1');
                if (saved) {
                    const data = JSON.parse(saved);
                    score = data.score || 0;
                    level = data.level || 1;
                    exp = data.exp || 0;
                    unlockedPrefs = new Set(data.unlockedPrefs || []);
                    unlockedBadges = new Set(data.unlockedBadges || []);
                    diffCorrectCount = data.diffCorrectCount || 0;
                }
            } catch(e) {}
            updateStatusUI();
        }

        function saveState() {
            try {
                const data = {
                    score,
                    level,
                    exp,
                    unlockedPrefs: Array.from(unlockedPrefs),
                    unlockedBadges: Array.from(unlockedBadges),
                    diffCorrectCount
                };
                localStorage.setItem('zengeos_save_v1', JSON.stringify(data));
            } catch(e) {}
        }

        function updateStatusUI() {
            document.getElementById('score-display').textContent = `${score} pts`;
            document.getElementById('player-level').textContent = `Lv.${level}`;
            
            const expNeeded = level * 100;
            const expPct = Math.min(100, Math.floor((exp / expNeeded) * 100));
            document.getElementById('exp-bar').style.width = `${expPct}%`;

            renderBadges();
        }

        function addExp(pts) {
            score += pts;
            exp += pts;
            const expNeeded = level * 100;
            if (exp >= expNeeded) {
                exp -= expNeeded;
                level++;
                playSound('levelup');
                confetti({ particleCount: 80, spread: 60, origin: { y: 0.6 } });
                checkAchievement('level_5');
            }
            updateStatusUI();
            saveState();
        }

        function unlockPrefecture(id) {
            if (!unlockedPrefs.has(id)) {
                unlockedPrefs.add(id);
                if (unlockedPrefs.size >= 10) checkAchievement('master_10');
                if (unlockedPrefs.size >= 47) checkAchievement('master_47');
                updateStatusUI();
                saveState();
            }
        }

        function checkAchievement(badgeId) {
            if (!unlockedBadges.has(badgeId)) {
                const def = achievementDefinitions.find(b => b.id === badgeId);
                if (def) {
                    unlockedBadges.add(badgeId);
                    playSound('levelup');
                    confetti({ particleCount: 70, spread: 50, origin: { y: 0.5 } });
                    showMapToast(`🏆 実績！【${def.name}】`, 'bg-amber-500 text-white');
                    updateStatusUI();
                    saveState();
                }
            }
        }

        function manualZoom(factor) {
            const svg = document.getElementById('japan-map-svg');
            const newW = currentViewBox.w / factor;
            const newH = currentViewBox.h / factor;

            if (newW < 120 || newW > 1200) return;

            const newX = currentViewBox.x + (currentViewBox.w - newW) / 2;
            const newY = currentViewBox.y + (currentViewBox.h - newH) / 2;

            currentViewBox = { x: newX, y: newY, w: newW, h: newH };
            svg.setAttribute('viewBox', `${newX} ${newY} ${newW} ${newH}`);
        }

        function startMapQuiz(mode) {
            mapQuizActive = true;
            mapQuizMode = mode;

            document.getElementById('map-start-btn').classList.add('hidden');
            document.getElementById('map-speed-btn').classList.add('hidden');

        function pickNextMapTarget() {
            const random = prefectures[Math.floor(Math.random() * prefectures.length)];
            mapTargetPref = random;

                ? `⚡ 「<span class="text-rose-600 font-black text-base underline decoration-wavy decoration-rose-400">${mapTargetPref.name}</span>」はどこ？`
                : `「<span class="text-emerald-600 font-black text-base underline decoration-wavy decoration-emerald-400">${mapTargetPref.name}</span>」をタップ！`;

            document.getElementById('map-target-prompt').innerHTML = promptText;
            document.getElementById('hint-box').classList.add('hidden');

            // Auto focus region
            let regionKey = 'ALL';
            if (['北海道', '東北'].includes(mapTargetPref.region)) regionKey = '東北';
            else if (['関東'].includes(mapTargetPref.region)) regionKey = '関東';
            else if (['中部'].includes(mapTargetPref.region)) regionKey = '中部';
            else if (['近畿'].includes(mapTargetPref.region)) regionKey = '近畿';
            else if (['中国', '四国'].includes(mapTargetPref.region)) regionKey = '中国';
            else if (['九州', '沖縄'].includes(mapTargetPref.region)) regionKey = '九州';

            const viewBoxStr = regionViewBoxes[regionKey];
            if (viewBoxStr) {
                const [x, y, w, h] = viewBoxStr.split(' ').map(Number);
                currentViewBox = { x, y, w, h };
                document.getElementById('japan-map-svg').setAttribute('viewBox', viewBoxStr);

        function handleMapClick(id) {
            const pref = prefectures.find(p => p.id === id);

            if (mapQuizActive && mapTargetPref) {
                const pathEl = document.getElementById(`pref-path-${id}`);
                if (id === mapTargetPref.id) {
                    playSound('correct');
                    checkAchievement('first_win');
                    unlockPrefecture(id);

                    if (mapQuizMode === 'speed') {
                        speedCorrectCount++;
                        addExp(40);
                        showMapToast(`⚡ 大正解！【${pref.name}】 (+40 EXP)`, 'bg-emerald-500 text-white');
                        if (speedCorrectCount >= 10) checkAchievement('speed_runner');
                    } else {
                        addExp(50);
                        showMapToast(`🎉 正解！【 ${pref.name} 】（県庁: ${pref.capital}） (+50 EXP)`, 'bg-emerald-500 text-white');
                    }

                    if (pathEl) pathEl.classList.add('correct-flash');

                    const txtEl = document.getElementById(`pref-text-${id}`);
                    if (txtEl) txtEl.textContent = pref.name.replace(/[都|府|県]/g, '');

                    setTimeout(() => {
                        if (pathEl) pathEl.classList.remove('correct-flash');
                        if (!showMapLabels && txtEl) txtEl.textContent = '';
                        pickNextMapTarget();
                    }, 900);
                } else {
                    playSound('wrong');
                    showMapToast(`❌ そこは【 ${pref.name} 】だよ！`, 'bg-rose-500 text-white');
                    if (pathEl) {
                        pathEl.classList.add('wrong-flash');
                        setTimeout(() => pathEl.classList.remove('wrong-flash'), 650);
                    }
                }
            } else {
                openModal(pref);
            }
        }

        // --- 4択クイズ LOGIC ---
        function setupNextQuiz() {
            document.getElementById('quiz-feedback-box').classList.add('hidden');

            const quizType = Math.floor(Math.random() * 4);
            const target = prefectures[Math.floor(Math.random() * prefectures.length)];
            currentQuizPref = target;

            const categoryBadge = document.getElementById('quiz-category-badge');
            const questionText = document.getElementById('quiz-question-text');
            const visualBox = document.getElementById('quiz-visual-box');
            const visualContent = document.getElementById('quiz-visual-content');

            visualBox.classList.add('hidden');

            let options = [target];
            while (options.length < 4) {
                const rand = prefectures[Math.floor(Math.random() * prefectures.length)];
                if (!options.find(o => o.id === rand.id)) options.push(rand);
            }
            options.sort(() => Math.random() - 0.5);

            if (quizType === 0) {
                categoryBadge.textContent = "🏢 県庁所在地クイズ";
                questionText.textContent = `「${target.name}」の県庁所在地はどこ？`;
                renderQuizOptions(options, target, 'capital');
            } else if (quizType === 1) {
                categoryBadge.textContent = "🌾 特産品・地理特徴";
                questionText.textContent = `「${target.goods.split(',')[0]}」などの特産を持つ都道府県は？`;
                visualBox.classList.remove('hidden');
                visualContent.textContent = target.name.charAt(0) + "❓";
                renderQuizOptions(options, target, 'name');
            } else if (quizType === 2) {
                const diffOnly = prefectures.filter(p => p.diff);
                const diffTarget = diffOnly[Math.floor(Math.random() * diffOnly.length)];
                currentQuizPref = diffTarget;
                categoryBadge.textContent = "⚠️ 重要！県名と異なる県庁所在地";
                questionText.textContent = `「${diffTarget.name}」の県庁所在地はどれ？`;
                
                let diffOptions = [diffTarget];
                while(diffOptions.length < 4) {
                    const r = prefectures[Math.floor(Math.random() * prefectures.length)];
                    if(!diffOptions.find(o => o.id === r.id)) diffOptions.push(r);
                }
                diffOptions.sort(() => Math.random() - 0.5);
                renderQuizOptions(diffOptions, diffTarget, 'capital');
            } else {
                categoryBadge.textContent = "🧩 ヒント・地方判定";
                questionText.textContent = `「${target.region}地方」で県庁が「${target.capital}」の都道府県は？`;
                renderQuizOptions(options, target, 'name');
            }

            const timerBar = document.getElementById('quiz-timer-bar');
            timerBar.style.width = '100%';
        }

        function renderQuizOptions(options, target, key) {
            const container = document.getElementById('quiz-options-container');
            container.innerHTML = '';

            options.forEach(opt => {
                const btn = document.createElement('button');
                btn.className = "w-full p-3 bg-white/90 active:bg-emerald-50 border border-slate-200 rounded-xl font-extrabold text-slate-800 text-xs text-left transition flex items-center justify-between shadow-xs";
                
                const displayText = key === 'capital' ? opt.capital : opt.name;
                btn.innerHTML = `
                    <span>${displayText}</span>
                    <i class="fa-solid fa-chevron-right text-slate-300 text-[10px]"></i>
                `;

                btn.onclick = () => checkQuizAnswer(opt, target, key);
                container.appendChild(btn);
            });
        }

        function checkQuizAnswer(selected, target, key) {
            const feedbackBox = document.getElementById('quiz-feedback-box');
            const feedbackText = document.getElementById('quiz-feedback-text');

            const isCorrect = selected.id === target.id;

            if (isCorrect) {
                playSound('correct');
                quizStreak++;
                unlockPrefecture(target.id);
                checkAchievement('first_win');

                if (quizStreak >= 5) checkAchievement('streak_5');
                if (quizStreak >= 10) checkAchievement('streak_10');

                if (target.diff) {
                    diffCorrectCount++;
                    if (diffCorrectCount >= 3) checkAchievement('diff_master');
                }

                const pts = 50 + (quizStreak * 10);
                addExp(pts);

                feedbackText.className = "text-xs font-black text-emerald-600";
                feedbackText.innerHTML = `🎉 大正解！ (${target.name} / 県庁: ${target.capital}) +${pts} EXP`;
            } else {
                playSound('wrong');
                quizStreak = 0;
                feedbackText.className = "text-xs font-black text-rose-500";
                const correctAns = key === 'capital' ? target.capital : target.name;
                feedbackText.innerHTML = `❌ 残念... 正解は「${correctAns}」！`;
            }

            document.getElementById('quiz-streak').textContent = `🔥 ${quizStreak}`;
            feedbackBox.classList.remove('hidden');
        }

        function renderBook() {
            const grid = document.getElementById('book-grid');
            const query = document.getElementById('book-search').value.toLowerCase();
            grid.innerHTML = '';

            const filtered = prefectures.filter(p => {
                const matchesRegion = currentRegionFilter === 'ALL' || p.region.includes(currentRegionFilter);
                const matchesSearch = p.name.includes(query) || p.capital.includes(query) || p.goods.includes(query);
                return matchesRegion && matchesSearch;
            });

            filtered.forEach(p => {
                const isUnlocked = unlockedPrefs.has(p.id);
                const card = document.createElement('div');
                card.className = "glass-panel p-3 rounded-2xl border border-white/90 active:scale-95 transition cursor-pointer flex flex-col justify-between space-y-1.5";
                card.onclick = () => openModal(p);

                card.innerHTML = `
                    <div>
                        <div class="flex justify-between items-center mb-0.5">
                            <span class="text-[9px] font-black px-1.5 py-0.2 rounded bg-sky-100 text-sky-700">${p.region}</span>
                            ${isUnlocked ? '<i class="fa-solid fa-circle-check text-emerald-500 text-xs"></i>' : ''}
                        </div>
                        <h4 class="text-sm font-black text-slate-800">${p.name}</h4>
                        <p class="text-[10px] font-extrabold text-emerald-600">県庁: ${p.capital}</p>
                    </div>
                    <div class="text-[9px] text-slate-500 bg-slate-50 p-1 rounded border border-slate-100 truncate">
                        ${p.goods}
                    </div>
                `;

                grid.appendChild(card);
            });
        }

                const isUnlocked = unlockedBadges.has(b.id);
                const card = document.createElement('div');
                card.className = `glass-panel p-3 rounded-2xl border text-center flex flex-col items-center justify-between space-y-1 ${isUnlocked ? 'bg-gradient-to-b from-white to-amber-50/60 border-amber-300' : 'badge-locked bg-slate-50'}`;

                card.innerHTML = `
                    <div class="text-2xl">${b.icon}</div>
                    <div>
                        <h4 class="text-xs font-black text-slate-800">${b.name}</h4>
                        <p class="text-[9px] font-bold text-slate-400 leading-tight mt-0.5">${b.desc}</p>
                    </div>
                    <span class="text-[9px] font-black px-2 py-0.5 rounded-full ${isUnlocked ? 'bg-amber-400 text-amber-950' : 'bg-slate-200 text-slate-500'}">
                        ${isUnlocked ? '獲得' : '未達'}
                    </span>
                `;

                grid.appendChild(card);
            });
        }

        // --- MODALS & TAB NAVIGATION ---
        function openModal(pref) {
            document.getElementById('modal-region').textContent = pref.region + "地方";
            document.getElementById('modal-title').textContent = pref.name;
            document.getElementById('modal-capital').textContent = pref.capital;
            document.getElementById('modal-diff').textContent = pref.diff ? "⚠️ 県名と異なる" : "✓ 県名と同じ";
            document.getElementById('modal-goods').textContent = pref.goods;
            document.getElementById('modal-desc').textContent = pref.desc;

            document.getElementById('pref-modal').classList.remove('hidden');
        }

        function closeModal() {
            document.getElementById('pref-modal').classList.add('hidden');
        }

        function switchTab(tab) {
            document.querySelectorAll('main > section').forEach(el => el.classList.add('hidden'));
            document.getElementById(`tab-${tab}`).classList.remove('hidden');

            document.querySelectorAll('.nav-btn').forEach(btn => {
                btn.classList.remove('text-emerald-700', 'bg-emerald-50', 'border', 'border-emerald-200');
                btn.classList.add('text-slate-500');
            });

            const activeBtn = document.getElementById(`nav-${tab}`);
            activeBtn.classList.remove('text-slate-500');
            activeBtn.classList.add('text-emerald-700', 'bg-emerald-50', 'border', 'border-emerald-200');

            if (tab === 'badges') renderBadges();
        }
